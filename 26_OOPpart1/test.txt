#include <iostream>
#include <vector>
#include <string>

using namespace std;

// 1. Описываем, что такое "Музыкальный трек"
// Это просто структура данных, которая хранит информацию об одной песне.
class MusicTrack {
public:
    string nameTrack = "Unknown";
    string artistTrack = "Unknown Artist";
    int trackLength = 0;
};

// 2. Описываем "Симулятор плеера"
// Он будет хранить список треков и управлять процессом воспроизведения.
class AudioPlayerSim {
public:
    // Наш плейлист — вектор, заполненный объектами MusicTrack
    vector<MusicTrack> playlist;

    // Переменная-курсор: какой трек сейчас выбран (0 — первый, 1 — второй и т.д.)
    int currentTrackIndex = 0;

    // Статус плеера
    bool isPlaying = false;

    // Метод для добавления новой песни в плеер
    // Мы передаем сюда готовый объект MusicTrack и кладем его в вектор playlist
    void addTrack(MusicTrack track) {
        playlist.push_back(track);
        cout << "Added: " << track.nameTrack << " to playlist." << endl;
    }

    // Метод воспроизведения
    void play() {
        if (playlist.empty()) {
            cout << "Playlist is empty! Add some tracks first." << endl;
            return;
        }

        // Берем трек из вектора по текущему индексу
        MusicTrack current = playlist[currentTrackIndex];

        cout << "\n--- NOW PLAYING ---" << endl;
        cout << "Track:  " << current.nameTrack << endl;
        cout << "Artist: " << current.artistTrack << endl;
        cout << "Length: " << current.trackLength << " sec" << endl;

        isPlaying = true;
    }

    // Метод переключения на следующий трек
    void next() {
        if (playlist.empty()) return;

        // Увеличиваем индекс на 1
        currentTrackIndex++;

        // Проверка: если индекс стал равен размеру списка (вышли за пределы),
        // сбрасываем его в 0, чтобы начать сначала (зацикливание)
        if (currentTrackIndex >= playlist.size()) {
            currentTrackIndex = 0;
        }

        cout << "\n[Next track selected]" << endl;
        play(); // Сразу запускаем воспроизведение нового трека
    }

    // Метод остановки
    void stop() {
        if (isPlaying) {
            cout << "\nStopping playback..." << endl;
            isPlaying = false;
        } else {
            cout << "\nPlayer is already stopped." << endl;
        }
    }
};

int main() {
    // Создаем объект плеера
    AudioPlayerSim myPlayer;

    // Создаем первый трек и заполняем его данные
    MusicTrack track1;
    track1.nameTrack = "In the End";
    track1.artistTrack = "Linkin Park";
    track1.trackLength = 216;

    // Создаем второй трек
    MusicTrack track2;
    track2.nameTrack = "Starboy";
    track2.artistTrack = "The Weeknd";
    track2.trackLength = 230;

    // Добавляем треки в плеер
    myPlayer.addTrack(track1);
    myPlayer.addTrack(track2);

    // --- ТЕСТИРУЕМ РАБОТУ ---

    myPlayer.play(); // Должен включиться Linkin Park

    myPlayer.next(); // Должен переключиться на The Weeknd

    myPlayer.next(); // Должен снова включить Linkin Park (так как сработало зацикливание)

    myPlayer.stop(); // Останавливаем

    return 0;
}
